USE tripdaobd;

-- ============================================================
-- TRIPDAOBD
-- Migration 003: Fix `users` table + create missing core tables
--
-- WHY THIS MIGRATION EXISTS
-- --------------------------------------------------------------
-- An audit of the backend found that several controllers already
-- query columns/tables that were never created by any existing
-- SQL file:
--
--   1. controllers/authController.js reads/writes
--      users.full_name, users.phone, users.profile_image,
--      users.is_verified — but database.sql only defines
--      users(id, name, email, password, created_at).
--      => registerUser() / loginUser() / updateProfile() would
--         throw "Unknown column" errors right now.
--
--   2. database/migrations/002_migrate_existing_users.sql reads
--      users.full_name and users.role, which also don't exist
--      yet — that migration cannot run successfully as-is.
--
--   3. controllers/notificationController.js,
--      bookingController.js and paymentController.js all
--      INSERT/SELECT against `notifications`, `bookings`, and
--      `payments` tables that have no CREATE TABLE statement
--      anywhere in the repo (payment.sql is just a sample SELECT,
--      not a schema file).
--
--   4. routes/bookingRoutes.js and adminBookingRoutes.js require
--      the permissions booking.view_all, booking.confirm,
--      booking.reject, booking.cancel_any — none of which were
--      inserted in 001_rbac_foundation.sql, so even an admin/
--      super_admin account could never pass those checks.
--
-- This migration fixes all four issues. Run it AFTER
-- 001_rbac_foundation.sql and BEFORE 002_migrate_existing_users.sql
-- (002 depends on users.full_name and users.role existing).
-- ============================================================


-- ------------------------------------------------------------
-- 1. FIX USERS TABLE
-- ------------------------------------------------------------

-- Add the columns the application code already expects.
ALTER TABLE users
  ADD COLUMN full_name VARCHAR(150) NULL AFTER id,
  ADD COLUMN phone VARCHAR(30) NULL AFTER email,
  ADD COLUMN profile_image VARCHAR(500) NULL AFTER password,
  ADD COLUMN is_verified BOOLEAN NOT NULL DEFAULT FALSE AFTER profile_image,
  ADD COLUMN role VARCHAR(30) NOT NULL DEFAULT 'user' AFTER is_verified;

-- Backfill full_name from the old `name` column so existing
-- accounts (if any) keep their data.
UPDATE users
SET full_name = name
WHERE full_name IS NULL;

-- full_name is required going forward.
ALTER TABLE users
  MODIFY COLUMN full_name VARCHAR(150) NOT NULL;

-- The old `name` column is superseded by `full_name` and is not
-- referenced anywhere in the current codebase. Safe to drop.
ALTER TABLE users
  DROP COLUMN name;


-- ------------------------------------------------------------
-- 2. NOTIFICATIONS TABLE
-- ------------------------------------------------------------

CREATE TABLE IF NOT EXISTS notifications (
    id INT NOT NULL AUTO_INCREMENT,
    user_id INT NOT NULL,

    title VARCHAR(200) NOT NULL,
    message VARCHAR(500) NOT NULL,
    type VARCHAR(50) NOT NULL DEFAULT 'system',
    link VARCHAR(255) NULL,

    is_read BOOLEAN NOT NULL DEFAULT FALSE,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    PRIMARY KEY (id),

    INDEX idx_notifications_user_id (user_id),
    INDEX idx_notifications_user_unread (user_id, is_read),

    CONSTRAINT fk_notifications_user
        FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);


-- ------------------------------------------------------------
-- 3. BOOKINGS TABLE
--    (columns match controllers/bookingController.js exactly)
-- ------------------------------------------------------------

CREATE TABLE IF NOT EXISTS bookings (
    id INT NOT NULL AUTO_INCREMENT,
    user_id INT NOT NULL,

    full_name VARCHAR(150) NOT NULL,
    email VARCHAR(150) NOT NULL,
    phone VARCHAR(30) NOT NULL,

    guests INT NOT NULL DEFAULT 1,
    rooms INT NOT NULL DEFAULT 1,

    check_in DATE NOT NULL,
    check_out DATE NOT NULL,

    special_request VARCHAR(500) NULL,

    destination VARCHAR(200) NOT NULL,
    location VARCHAR(200) NOT NULL,

    room_price DECIMAL(10,2) NOT NULL DEFAULT 0,
    service_fee DECIMAL(10,2) NOT NULL DEFAULT 0,
    vat DECIMAL(10,2) NOT NULL DEFAULT 0,
    total_price DECIMAL(10,2) NOT NULL DEFAULT 0,

    status ENUM(
        'Pending',
        'Confirmed',
        'In Progress',
        'Completed',
        'Cancelled',
        'Rejected'
    ) NOT NULL DEFAULT 'Pending',

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    PRIMARY KEY (id),

    INDEX idx_bookings_user_id (user_id),
    INDEX idx_bookings_status (status),

    CONSTRAINT fk_bookings_user
        FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);


-- ------------------------------------------------------------
-- 4. PAYMENTS TABLE
--    (columns match controllers/paymentController.js exactly)
-- ------------------------------------------------------------

CREATE TABLE IF NOT EXISTS payments (
    id INT NOT NULL AUTO_INCREMENT,
    booking_id INT NOT NULL,
    user_id INT NOT NULL,

    amount DECIMAL(10,2) NOT NULL,

    payment_method ENUM(
        'bkash',
        'nagad',
        'card',
        'bank'
    ) NOT NULL,

    transaction_id VARCHAR(100) NOT NULL,

    payment_status ENUM(
        'Pending',
        'Paid',
        'Failed',
        'Refunded',
        'Partially Refunded'
    ) NOT NULL DEFAULT 'Pending',

    paid_at TIMESTAMP NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    PRIMARY KEY (id),

    UNIQUE KEY uk_payments_transaction_id (transaction_id),

    INDEX idx_payments_booking_id (booking_id),
    INDEX idx_payments_user_id (user_id),

    CONSTRAINT fk_payments_booking
        FOREIGN KEY (booking_id)
        REFERENCES bookings(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE,

    CONSTRAINT fk_payments_user
        FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);


-- ------------------------------------------------------------
-- 5. MISSING ADMIN BOOKING PERMISSIONS
--
--    routes/bookingRoutes.js and adminBookingRoutes.js already
--    require these permission names, but 001_rbac_foundation.sql
--    never created them — so no role (not even admin) could ever
--    pass those route checks.
-- ------------------------------------------------------------

INSERT IGNORE INTO permissions (name, description)
VALUES
    ('booking.view_all', 'View all bookings across the platform'),
    ('booking.confirm', 'Confirm a pending booking'),
    ('booking.reject', 'Reject a pending booking'),
    ('booking.cancel_any', 'Cancel any booking regardless of owner');

-- Grant them to the admin role (super_admin already receives
-- every permission automatically via its CROSS JOIN rule in
-- 001_rbac_foundation.sql, so no change needed there).
INSERT IGNORE INTO role_permissions (role_id, permission_id)
SELECT
    r.id,
    p.id
FROM roles r
JOIN permissions p
WHERE r.name = 'admin'
AND p.name IN (
    'booking.view_all',
    'booking.confirm',
    'booking.reject',
    'booking.cancel_any'
);
